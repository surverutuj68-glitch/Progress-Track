268. Missing Number

class Solution {
    public int missingNumber(int[] nums) {
        List<Integer> li = new ArrayList<>();
        for(int num : nums){
            li.add(num);
        }
        for(int i=0;i<= nums.length;i++){
            if(!li.contains(i)){
               return i;
            }
        }
        return 0;
    }
}

time complexity - O(N^2)
space complexity - O(N)

class Solution {
    public int missingNumber(int[] nums) {
        int n = nums.length;
        int expected = n*(n+1)/2;
        int actual = 0;
        for(int i=0;i<nums.length;i++){
            actual += nums[i];
        }
        return expected - actual;
    }
}
time complexity - O(N)
space complexity - O(1)
