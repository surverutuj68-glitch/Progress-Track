9. Palindrome Number

class Solution {
    public boolean isPalindrome(int x) {
        String a = Integer.toString(x);
        for(int i=0;i<a.length()/2;i++){
            if(a.charAt(i) != a.charAt(a.length()-1-i)){
                return false;
            }
        }
        return true;
    }
}
