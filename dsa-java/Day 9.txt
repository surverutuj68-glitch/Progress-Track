283. Move Zeroes

class Solution {
    public void moveZeroes(int[] nums) {
        int pos=0;
        for(int i =0;i<nums.length;i++){
            if(nums[i]!=0){
                int a = nums[i];
                nums[i]=nums[pos];
                nums[pos]=a;
                pos++;
            }
        }
    }
}

time complexity - O(n)
space complexity - O(1)
