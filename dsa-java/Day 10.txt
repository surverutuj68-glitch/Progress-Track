53. Maximum Subarray

class Solution {
    public int maxSubArray(int[] nums) {
        int max = nums[0];
        int curr = nums[0];
        for(int i=1;i<nums.length;i++){
            curr = Math.max(nums[i],curr+nums[i]);
            if(curr>max){
                max=curr;
            }
        }
        return max;
    }
}

time complexity - O(n)
space complexity - O(1)
