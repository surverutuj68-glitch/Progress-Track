199. Binary Tree Right Side View

class Solution {
    List<Integer> li = new ArrayList<>();
    public List<Integer> rightSideView(TreeNode root) {
        Queue<TreeNode> q = new LinkedList<>();
        if(root == null){
            return li;
        }
        q.offer(root);
        while(!q.isEmpty()){
            int size = q.size();

            for(int i = 0; i < size; i++){

                TreeNode node = q.poll();
                if(i == size - 1){
                    li.add(node.val);
                }

                if(node.left != null){
                    q.offer(node.left);
                }

                if(node.right != null){
                    q.offer(node.right);
                }
            }
        }
        return li;
        }
    }

time complexity - O(N)
space complexity - O(N)
