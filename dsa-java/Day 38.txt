142. Linked List Cycle II

public class Solution {
    public ListNode detectCycle(ListNode head) {
        ListNode curr = head;
        HashSet<ListNode> hs = new HashSet<>();
        while(curr!=null){
            if(hs.contains(curr)){
                return curr;
            }
            hs.add(curr);
            curr = curr.next;
        }
        return null;
    }
}

time complexity - O(N)
space complexity - O(N)
